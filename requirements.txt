# ===== Web 框架 =====
# FastAPI:现代 Python Web 框架,自带数据校验和自动生成 API 文档
fastapi==0.115.5
# uvicorn:运行 FastAPI 的 ASGI 服务器(相当于前端的 node 运行时)
# [standard] 会额外装上性能相关的依赖(如 uvloop、httptools)
uvicorn[standard]==0.32.1

# ===== 配置与数据校验 =====
# pydantic:定义数据结构并自动校验(类比 TS 的 interface + zod)
pydantic==2.10.3
# pydantic-settings:从环境变量 / .env 文件读取配置
pydantic-settings==2.6.1

# ===== 大模型调用 =====
# openai:官方 SDK。它不只能连 OpenAI——只要服务商提供"OpenAI 兼容接口"
# (通义千问、DeepSeek、Kimi 等都提供),改一下 base_url 就能直接用同一套代码。
openai==1.99.9

# ===== 数据库(PostgreSQL)=====
# SQLModel:FastAPI 作者出品,把 Pydantic 和 SQLAlchemy 融合。
# 好处是"一个类既能当 API 请求校验,又能当数据库表",学习曲线最平。
# 底层其实就是 SQLAlchemy 2.x,以后想切纯 SQLAlchemy 也几乎无痛。
sqlmodel==0.0.22
# psycopg 3:PostgreSQL 官方推荐的 Python 驱动(同步模式)。
# 与 FastAPI 官方全栈模板同款选型——业务层全是同步 Session,换库不动业务代码。
# [binary] 附带预编译二进制包,免本地编译。
psycopg[binary]==3.3.5
# Alembic:版本化管理数据库结构,生产环境不依赖 create_all() 自动建表
alembic==1.14.1
# pgvector:PG 向量列的 SQLAlchemy 类型 + python 侧向量序列化。
# 语料向量检索(guide_chunks 表)依赖它;纯 python,依赖 numpy,无编译步骤。
pgvector==0.4.2

# 密码加盐
passlib==1.7.4
# bcrypt 4.1.0+ 删除了 __about__ 模块,passlib 1.7.4 读取它会报
# AttributeError: module 'bcrypt' has no attribute '__about__',故锁到 4.0.x
bcrypt==4.0.1

# ===== 登录鉴权 =====
# PyJWT:签发/校验 JWT token。登录成功后发一个 token 给前端,
# 之后每个请求带 Authorization: Bearer <token>,服务端验签就知道是谁,
# 不用维护 session 状态。
PyJWT==2.10.1
# python-multipart:FastAPI 的 OAuth2PasswordBearer 依赖它解析表单,
# 装上避免运行时报 "Form data requires python-multipart"。
python-multipart==0.0.20

# ===== 缓存 =====
# redis:官方 Python 客户端(redis-py)。客户端内部自带连接池且线程安全,
# 所以全局建一个实例复用即可,不要每次请求 new 一个。
# 4.2 起已内置了原 aioredis 的异步能力,不需要额外装异步库。
redis==5.2.1

# ===== HTTP 客户端与外部检索(RAG 模块)=====
# 消费者:app/services/rag/(page_fetcher 抓页、wiki_provider 调 MediaWiki API、
# search_provider 的 DDG 兜底实现)。
# httpx:发送 HTTP 请求。与测试里 TestClient 底层同源,API 习惯一致。
httpx==0.28.1
# ddgs:DuckDuckGo 搜索(原 duckduckgo_search 更名而来),免注册、无 API Key。
# 现在是**兜底** Provider(settings.search_provider="ddg"),默认走 MediaWiki——
# 实测 DDG 连续请求会被限流且不恢复,详见 docs/notes/18-rag-agent-and-retrieval.md。
ddgs==9.16.0
