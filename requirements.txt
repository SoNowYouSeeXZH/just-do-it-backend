# ===== Web 框架 =====
# FastAPI:现代 Python Web 框架,自带数据校验和自动生成 API 文档
fastapi==0.115.5
# uvicorn:运行 FastAPI 的 ASGI 服务器(相当于前端的 node 运行时)
# [standard] 会额外装上性能相关的依赖(如 uvloop、httptools)
uvicorn[standard]==0.32.1

# ===== 配置与数据校验 =====
# pydantic:定义数据结构并自动校验(类比 TS 的 interface + zod)
pydantic==2.10.3
# pydantic-settings:从环境变量 / .env 文件读取配置
pydantic-settings==2.6.1

# ===== 大模型调用 =====
# openai:官方 SDK。它不只能连 OpenAI——只要服务商提供"OpenAI 兼容接口"
# (通义千问、DeepSeek、Kimi 等都提供),改一下 base_url 就能直接用同一套代码。
openai==1.99.9

# ===== 数据库(MySQL)=====
# SQLModel:FastAPI 作者出品,把 Pydantic 和 SQLAlchemy 融合。
# 好处是"一个类既能当 API 请求校验,又能当数据库表",学习曲线最平。
# 底层其实就是 SQLAlchemy 2.x,以后想切纯 SQLAlchemy 也几乎无痛。
sqlmodel==0.0.22
# PyMySQL:纯 Python 写的 MySQL 驱动,免编译、跨平台。
# 我们先用同步驱动,代码好读;将来要吞吐再换 asyncmy(异步驱动)。
PyMySQL==1.1.1
# Alembic:版本化管理数据库结构,生产环境不依赖 create_all() 自动建表
alembic==1.14.1
# cryptography:MySQL 8 默认用 caching_sha2_password 认证,
# PyMySQL 必须借助 cryptography 才能算出加密握手,否则报
# "'cryptography' package is required for ... auth methods"。
cryptography==43.0.3

# 密码加盐
passlib==1.7.4
# bcrypt 4.1.0+ 删除了 __about__ 模块,passlib 1.7.4 读取它会报
# AttributeError: module 'bcrypt' has no attribute '__about__',故锁到 4.0.x
bcrypt==4.0.1

# ===== 登录鉴权 =====
# PyJWT:签发/校验 JWT token。登录成功后发一个 token 给前端,
# 之后每个请求带 Authorization: Bearer <token>,服务端验签就知道是谁,
# 不用维护 session 状态。
PyJWT==2.10.1
# python-multipart:FastAPI 的 OAuth2PasswordBearer 依赖它解析表单,
# 装上避免运行时报 "Form data requires python-multipart"。
python-multipart==0.0.20
